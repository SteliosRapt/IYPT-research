% --- Figures ---
% \fig{filename}{caption}{label} - Simple centered figure
\newcommand{\fig}[3]{%
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.8\textwidth]{#1}
        \caption{#2}
        \label{fig:#3}
    \end{figure}
}

% \figw{filename}{width}{caption}{label} - Figure with custom width
\newcommand{\figw}[4]{%
    \begin{figure}[H]
        \centering
        \includegraphics[width=#2\textwidth]{#1}
        \caption{#3}
        \label{fig:#4}
    \end{figure}
}
